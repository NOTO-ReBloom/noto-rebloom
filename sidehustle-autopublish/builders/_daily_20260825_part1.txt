def enc_json(obj):
    plain=json.dumps(obj,ensure_ascii=False,separators=(",",":")).encode(); key=os.urandom(32); iv=os.urandom(12)
    sealed=AESGCM(key).encrypt(iv,plain,None); ct,tag=sealed[:-16],sealed[-16:]
    wrapped=pub.encrypt(key,asym_padding.OAEP(mgf=asym_padding.MGF1(hashes.SHA256()),algorithm=hashes.SHA256(),label=None))
    env={"version":1,"alg":"AES-256-GCM+RSA-OAEP-SHA256","wrappedKey":base64.b64encode(wrapped).decode(),"iv":base64.b64encode(iv).decode(),"tag":base64.b64encode(tag).decode(),"ciphertext":base64.b64encode(ct).decode()}
    return json.dumps(env,separators=(",",":")).encode()
